    %% ============================================================================
    %% CORE PACKAGE - Central logging interfaces and orchestration
    %% ============================================================================
    namespace Core {
        class LogLevel {
            <<enumeration>>
            DEBUG
            INFO
            WARNING
            ERROR
            FATAL
            IsGreaterOrEqual(LogLevel) bool$
        }

        class LogMessage {
            <<valueObject>>
            +Timestamp DateTime
            +LogLevel LogLevel
            +Message string
            +Source string
            +ThreadId int
            +CorrelationId string
        }

        class ISmartLogger {
            <<interface>>
            +Log(LogLevel, string) void*
            +Debug(string) void*
            +Info(string) void*
            +Warning(string) void*
            +Error(string) void*
            +Fatal(string) void*
            +SetLogLevel(LogLevel) void*
            +AddAppender(ILogAppender) void*
        }

        class LoggerImplementation {
            <<internal>>
            -filters: ILogFilter[]
            -appenders: ILogAppender[]
            -logLevel: LogLevel
            +Log(LogLevel, string) void
            +SetLogLevel(LogLevel) void
            +AddAppender(ILogAppender) void
        }

        class ILogAppender {
            <<interface>>
            +Append(LogMessage) void*
            +SetLogLevel(LogLevel) void*
            +GetLogLevel(LogLevel) LogLevel*
            +IsEnabled(LogLevel) bool*
            +SetFormatter(ILogOutputFormatterStrategy) void*
            +GetFormatter() ILogOutputFormatterStrategy*
        }

        class ILogConfigurationProvider {
            <<interface>>
            +Load() LogConfigurationHolder*
        }

        class ILogFilter {
            <<interface>>
            +ShouldLog(LogMessage) bool*
            +SetLogLevel(LogLevel) void*
            +GetLogLevel() LogLevel*
        }

        class ILogLayoutStrategy {
            <<interface>>
            +Render(LogMessage) string*
        }

        class ILogOutputFormatterStrategy {
            <<interface>>
            +Format(LogMessage) string*
        }

        class ITokenRendererStrategy {
            <<interface>>
            +Token string*
            +Render(LogMessage) string*
        }

        class LogConfigurationHolder {
            <<internal>>
            -appenderConfigs: AppenderConfiguration[]
            -filters: ILogFilter[]
            -globalLogLevel: LogLevel
        }

        class LoggerFactory {
            <<internal>>
            -configuration: LogConfigurationHolder
            -loggers: ConcurrentDictionary~string, ISmartLogger~
            +GetOrCreateLogger(string) ISmartLogger
            +ReloadConfiguration(LogConfigurationHolder) void
        }

        class LoggerManager {
            <<static>>
            +Initialize(ILogConfigurationProvider) void$
            +GetLogger(string) ISmartLogger$
            +GetLogger(Type) ISmartLogger$
            +ReloadConfiguration(ILogConfigurationProvider) void$
        }

        class LogContext {
            <<static>>
            +GetCorrelationId() string$
            +SetCorrelationId(string) void$
            +ClearCorrelationId() void$
        }
    }

    %% ============================================================================
    %% APPENDERS PACKAGE - Output destinations
    %% ============================================================================
    namespace Appenders {
        class ConsoleAppender {
            <<sealed>>
            -logLevel: LogLevel
            -formatter: ILogOutputFormatterStrategy
            -lockObject: object
            +Append(LogMessage) void
            +SetLogLevel(LogLevel) void
            +IsEnabled(LogLevel) bool
            +SetFormatter(ILogOutputFormatterStrategy) void
        }

        class FileAppender {
            <<sealed>>
            -logLevel: LogLevel
            -formatter: ILogOutputFormatterStrategy
            -fileNameBuilder: FileNameBuilder
            -rollingStrategy: IRollingStrategy
            -lockObject: object
            +Append(LogMessage) void
            +SetLogLevel(LogLevel) void
            +IsEnabled(LogLevel) bool
            +SetFormatter(ILogOutputFormatterStrategy) void
        }

        class AppenderConfiguration {
            +Destination: LogOutputDestination
            +OutputFormat: LogOutputFormat
            +LayoutType: LogMessageLayoutType
            +JsonFields: string[]
            +JsonFieldMapping: Dictionary~string, string~
        }

        class LogOutputDestination {
            <<enumeration>>
            Console
            File
        }

        class LogOutputFormat {
            <<enumeration>>
            PlainText
            Json
            Xml
        }

        class LogMessageLayoutType {
            <<enumeration>>
            Simple
            Pattern
        }
    }

    %% ============================================================================
    %% FILE ROLLING PACKAGE - File rolling strategies
    %% ============================================================================
    namespace FileRolling {
        class IRollingStrategy {
            <<interface>>
            +ShouldRoll(string) bool*
            +GetNextFilePath(string) string*
            +OnRoll(string) void*
        }

        class SizeRollingStrategy {
            <<sealed>>
            -maxFileSizeBytes: long
            +ShouldRoll(string) bool
            +GetNextFilePath(string) string
        }

        class TimeRollingStrategy {
            <<sealed>>
            -rollingInterval: TimeSpan
            -lastRollTime: DateTime
            +ShouldRoll(string) bool
            +GetNextFilePath(string) string
        }

        class FileNameBuilder {
            <<internal>>
            -fileConfig: FileConfiguration
            +Build() string
        }

        class FileConfiguration {
            +FilePath: string
            +FileName: string
            +FileExtension: string
            +MaxFileSizeBytes: long
            +RollingInterval: TimeSpan
            +RollingStrategy: RollingStrategyType
        }

        class RollingStrategyType {
            <<enumeration>>
            None
            Size
            Time
        }

        class RollingFactory {
            +Create(FileConfiguration) IRollingStrategy*
        }
    }

    %% ============================================================================
    %% CONFIGURATIONS PACKAGE - Configuration providers
    %% ============================================================================
    namespace Configurations {
        class JsonConfigurationProvider {
            +Load() LogConfigurationHolder
            +EnableAutoReload() void
            -ParseJsonConfiguration(string) LogConfigurationHolder
        }

        class InMemoryConfigurationProvider {
            -configuration: LogConfigurationHolder
            +Load() LogConfigurationHolder
            +SetConfiguration(LogConfigurationHolder) void
        }
    }

    %% ============================================================================
    %% FORMATTERS PACKAGE - Output formatting
    %% ============================================================================
    namespace Formatters {
        class FormatterFactory {
            <<static>>
            +Create(AppenderConfiguration) ILogOutputFormatterStrategy$
        }

        class PlainTextFormatter {
            -layout: ILogLayoutStrategy
            +Format(LogMessage) string
        }

        class JsonFormatter {
            -jsonFields: string[]
            -fieldMapping: Dictionary~string, string~
            +Format(LogMessage) string
        }

        class XmlFormatter {
            +Format(LogMessage) string
        }
    }

    %% ============================================================================
    %% LOG LAYOUTS PACKAGE - Pattern-based layout rendering
    %% ============================================================================
    namespace LogLayouts {
        class LayoutFactory {
            <<static>>
            +Create(AppenderConfiguration) ILogLayoutStrategy$
        }

        class PatternLayout {
            -pattern: string
            -tokenRegistry: TokenRegistry
            +Render(LogMessage) string
        }
    }

    %% ============================================================================
    %% TOKENS PACKAGE - Token rendering strategies
    %% ============================================================================
    namespace Tokens {
        class TokenRegistry {
            <<internal>>
            -tokens: Dictionary~string, ITokenRendererStrategy~
            +Tokens Dictionary*
        }

        class TimestampToken {
            <<sealed>>
            +Token string
            +Render(LogMessage) string
        }

        class LevelToken {
            <<sealed>>
            +Token string
            +Render(LogMessage) string
        }

        class MessageToken {
            <<sealed>>
            +Token string
            +Render(LogMessage) string
        }

        class SourceToken {
            <<sealed>>
            +Token string
            +Render(LogMessage) string
        }

        class ThreadToken {
            <<sealed>>
            +Token string
            +Render(LogMessage) string
        }

        class CorrelationToken {
            <<sealed>>
            +Token string
            +Render(LogMessage) string
        }
    }

    %% ============================================================================
    %% FILTERS PACKAGE - Log filtering
    %% ============================================================================
    namespace Filters {
        class LevelFilter {
            <<sealed>>
            +ShouldLog(LogMessage) bool
            +SetLogLevel(LogLevel) void
            +GetLogLevel() LogLevel
        }
    }
